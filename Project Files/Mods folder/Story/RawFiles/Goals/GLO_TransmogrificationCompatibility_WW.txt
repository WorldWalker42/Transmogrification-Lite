//REGION Handle mutually exclusive armor buff passives

// Give the new passive
IF
DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff((GUIDSTRING)_Character,(STRING)_Passive)
THEN
AddPassive(_Character,_Passive);

// Remove the previous passive if there is one
IF
DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff(_Character,_Passive)
AND
DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff(_Character,_OtherPassive)
AND
_Passive != _OtherPassive
THEN
NOT DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff(_Character,_OtherPassive);
RemovePassive(_Character,_OtherPassive);

// Easy way to clear current passive
PROC
PROC_WW_TL_Compatibility_RemoveMutuallyExclusivePassive_ArmorBuff((GUIDSTRING)_Character)
AND
DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff(_Character,_Passive)
THEN
NOT DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff(_Character,_Passive);
RemovePassive(_Character,_Passive);

// Automatically clear current passive if character unequips their last transmog
PROC
PROC_WW_TL_Event_Unequipped((ITEM)_,(CHARACTER)_Character)
AND
NOT DB_WW_TL_EquippedTransmogrifications(_Character,(ITEM)_)
AND
DB_WW_TL_Compatibility_HasMutuallyExclusivePassive_ArmorBuff(_Character,_)
THEN
PROC_WW_TL_Compatibility_RemoveMutuallyExclusivePassive_ArmorBuff(_Character);

//END_REGION

//REGION Triggers and general logic for special mod compatibility

PROC
PROC_WW_TL_Event_Equipped((ITEM)_,(CHARACTER)_Character)
THEN
PROC_WW_TL_CompatibilityCheck(_Character);

PROC
PROC_WW_TL_Event_Unequipped((ITEM)_,(CHARACTER)_Character)
THEN
PROC_WW_TL_CompatibilityCheck(_Character);

// Custom trigger for equipping shield with any transmog equipped
IF
Equipped(_Shield,_Character)
AND
DB_WW_TL_EquippedTransmogrifications((GUIDSTRING)_Character,(ITEM)_)
AND
IsTagged((GUIDSTRING)_Shield,(TAG)WPN_SHIELD_dc9e09f0-b2ba-4720-bd28-ee9ddbdef2f1,1)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",500);

// Custom trigger for unequipping shield
IF
Unequipped(_Shield,_Character)
AND
DB_WW_TL_EquippedTransmogrifications((GUIDSTRING)_Character,(ITEM)_)
AND
IsTagged((GUIDSTRING)_Shield,(TAG)WPN_SHIELD_dc9e09f0-b2ba-4720-bd28-ee9ddbdef2f1,1)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",500);

// Custom trigger for gaining unarmored-only status
IF
StatusApplied(_Character,_Status,_,_)
AND
DB_WW_TL_UnarmoredOnlyStatuses(_Status)
AND
DB_WW_TL_EquippedTransmogrifications(_Character,_)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",500);

// Custom trigger for losing unarmored-only status
IF
StatusRemoved(_Character,_Status,_,_)
AND
DB_WW_TL_UnarmoredOnlyStatuses(_Status)
AND
DB_WW_TL_EquippedTransmogrifications(_Character,_)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",500);

// Custom trigger for leveling up (will probably need to update AC after respeccing, multiclassing, or gaining new class features)
IF
LeveledUp(_Character)
AND
DB_WW_TL_EquippedTransmogrifications((GUIDSTRING)_Character,_)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",750);

IF
ObjectTimerFinished(_Character,"WW_TL_Compatibility_QueueCheck")
THEN
PROC_WW_TL_CompatibilityCheck((CHARACTER)_Character);

IF
ObjectTimerFinished(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC")
THEN
PROC_WW_TL_CompatibilityCheck((CHARACTER)_Character);
RealtimeObjectTimerLaunch(_Character,"WW_TL_TriggerACReevaluation",500); // will trigger rule in main script

//REGION Apply/remove standard armor tracker passive (and logic related to it)

// Check whether character needs this passive
PROC
PROC_WW_TL_CompatibilityCheck((CHARACTER)_Character)
AND
HasPassive((GUIDSTRING)_Character,"WW_TL_Compatibility_StandardArmorTracker",0)
AND
QRY_WW_TL_Compatibility_NeedsStandardArmorTracker(_Character)
THEN
AddPassive(_Character,"WW_TL_Compatibility_StandardArmorTracker");
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck",500); // queue another compatibility check after passive has time to resolve (it might apply armor status)

// Check whether character no longer needs this passive
PROC
PROC_WW_TL_CompatibilityCheck((CHARACTER)_Character)
AND
HasPassive((GUIDSTRING)_Character,"WW_TL_Compatibility_StandardArmorTracker",1)
AND
NOT QRY_WW_TL_Compatibility_NeedsStandardArmorTracker(_Character)
THEN
RemovePassive(_Character,"WW_TL_Compatibility_StandardArmorTracker");
RemoveStatus(_Character,"WW_TL_COMPATIBILITY_WEARING_NORMAL_ARMOR"); // just in case this status is currently applied

IF
StatusApplied(_Character,"WW_TL_COMPATIBILITY_WEARING_NORMAL_ARMOR",_,_)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",250);

IF
StatusRemoved(_Character,"WW_TL_COMPATIBILITY_WEARING_NORMAL_ARMOR",_,_)
THEN
RealtimeObjectTimerLaunch(_Character,"WW_TL_Compatibility_QueueCheck_IncludeAC",250);

//END_REGION

//REGION Poverty Domain subclass by TheWailingBard - https://mod.io/g/baldursgate3/m/poverty-domain

// This one's complicated:
// IF NOT EQUIPPED any armor or certain types of weapons (only clothes transmog allowed), set override and alt AC tags but otherwise do nothing (actual override is active)
// IF EQUIPPED only transmogrified armor, set override and alt AC tags, then set passives to undo WIS + CON, and finally add appropriate amount of DEX (actual override is active)
//    * Add full DEX if wearing no transmog on chest and transmog armor on another slot
//    * Add full DEX if wearing transmog clothes or light armor on chest
//    * Add partial DEX if wearing transmog medium armor on chest
// IF EQUIPPED any actual armor or certain types of weapons, use normal AC calculations (actual override is inactive)
// IF DEX >= WIS + CON, need to also undo WIS + CON and add full DEX when wearing transmog clothes on chest (actual override is still active when it shouldn't be when equipped transmog)
