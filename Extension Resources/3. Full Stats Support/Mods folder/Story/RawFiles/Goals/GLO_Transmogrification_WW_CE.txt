// The INIT section of the transmogrification compatibility script must at least add a fact to DB_WW_TL_ArmorComponents for each piece of equipment.
// The database has the following format:
// DB_WW_TL_ArmorComponents(_Template, _AppearanceTemplate, _StatsStatusName, _ArmorType, _DisplayAC, _Slot, _Unique, _AdditionalAttributesTooltip);

// In this example, the following items are fully supported for taking their appearance AND stats:
DB_WW_TL_ArmorComponents(WW_CE_ARM_Breastplate_WalkersPlate_Damaged_3e817f24-38d7-4f18-827e-531a220fd4a9, WW_CE_ARM_Breastplate_WalkersPlate_Damaged_AppearanceOnly_3d98ad35-afce-472f-a393-a0129bd4345e, "WW_CE_TL_ARM_Breastplate_WalkersPlate_Damaged", 1, 12, "Breast", 0, 1);
DB_WW_TL_ArmorComponents(WW_CE_ARM_Breastplate_WalkersPlate_c97df1d0-3011-47f4-b859-3ea9e3664997, WW_CE_ARM_Breastplate_WalkersPlate_AppearanceOnly_fa942a7b-ea01-4562-be3c-19854ae8a255, "WW_CE_TL_ARM_Breastplate_WalkersPlate", 5, 14, "Breast", 1, 1);
DB_WW_TL_ArmorComponents(WW_CE_ARM_Boots_WalkersPlate_39001a4c-8a06-47d9-8e22-5b5a9b545a5b, WW_CE_ARM_Boots_WalkersPlate_AppearanceOnly_b3acf254-166c-4bab-9b36-3503a046dd3b, "WW_CE_TL_ARM_Boots_WalkersPlate", 0, 0, "Boots", 1, 0);

// One item unlocks a spell with a Short Rest cooldown that needs to be enforced:
DB_WW_TL_ArmorSpells("WW_CE_TL_ARM_Boots_WalkersPlate", "Projectile_WW_CE_Fly", "SHORT");

// This item's stats can also be described with the following tooltip statuses:
DB_WW_TL_TooltipStatuses("WW_CE_TL_ARM_Boots_WalkersPlate", "SPELL");

KBSECTION
// The KB section of the transmogrification compatibility script can optionally be used to implement / enforce more advanced armor behaviors.

// For example, the undamaged breastplate shouldn't be able to be equipped by anyone without the AVATAR tag, even when it's been transmogrified to look different.
// So, if it's equipped by someone without the tag, this rule will automatically unequip it and open a popup message that tells the player what happened.
IF
Equipped(_Item, _Character)
AND
DB_WW_TL_TransmogrifiedArmors((GUIDSTRING)_Item, (INTEGER)_, WW_CE_ARM_Breastplate_WalkersPlate_c97df1d0-3011-47f4-b859-3ea9e3664997, (STRING)_)
AND
IsTagged((GUIDSTRING)_Character, (TAG)AVATAR_306b9b05-1057-4770-aa17-01af21acd650, 0)
AND
ResolveTranslatedString("WW_CE_AvatarOnlyErrorMessage", _ErrorMessage)
THEN
Unequip((CHARACTER)_Character, (ITEM)_Item);
OpenMessageBox(_Character, _ErrorMessage);

//REGION Set Bonus

// If a character equips an original item from the set, call the PROC to evaluate the set bonus
IF
TemplateEquipped(_Template, _Character)
AND
DB_WW_CE_SetBonusItems(_Template)
THEN
PROC_WW_CE_EvaluateSetBonus(_Character);

// Or, if a character equips any transmogrification, also call the PROC to evaluate the set bonus
IF
DB_WW_TL_EquippedTransmogrifications((GUIDSTRING)_Character, (ITEM)_)
THEN
PROC_WW_CE_EvaluateSetBonus((CHARACTER)_Character);

// If _Character has the entire set equipped and doesn't already have the set bonus, this PROC will apply it
PROC
PROC_WW_CE_EvaluateSetBonus((CHARACTER)_Character)
AND
QRY_WW_CE_EntireSetEquipped(_Character)
AND
HasActiveStatus((GUIDSTRING)_Character, "WW_CE_PLANE_WALKERS_SET_BONUS", 0)
THEN
ApplyStatus(_Character, "WW_CE_PLANE_WALKERS_SET_BONUS", -1.0);

// If a character unequips any item and doesn't have the entire set equipped, remove the set bonus if they currently have it
IF
Unequipped(_, _Character)
AND
NOT QRY_WW_CE_EntireSetEquipped(_Character)
AND
HasActiveStatus((GUIDSTRING)_Character, "WW_CE_PLANE_WALKERS_SET_BONUS", 1)
THEN
RemoveStatus(_Character, "WW_CE_PLANE_WALKERS_SET_BONUS");

// A custom query that evaluates to true if the character is using the stats for every piece of equipment in the set
QRY
QRY_WW_CE_EntireSetEquipped((CHARACTER)_Character)
AND
QRY_WW_CE_UsingStatsFromTemplate(_Character, WW_CE_ARM_Breastplate_WalkersPlate_c97df1d0-3011-47f4-b859-3ea9e3664997)
AND
QRY_WW_CE_UsingStatsFromTemplate(_Character, WW_CE_ARM_Boots_WalkersPlate_39001a4c-8a06-47d9-8e22-5b5a9b545a5b)
THEN
DB_NOOP(1);

// The first part of a custom query, which will make it evaluate to true if the character has equipped an item made from the original equipment template.
QRY
QRY_WW_CE_UsingStatsFromTemplate((CHARACTER)_Character, (GUIDSTRING)_Template)
AND
DB_WW_TL_ArmorComponents(_Template, _, _, _, _, _Slotname, _, _)
AND
GetEquippedItem(_Character, _Slotname, _Item)
AND
GetTemplate(_Item, _Template)
THEN
DB_NOOP(1);

// The second part of a custom query, which will make it evaluate to true if the character has equipped a transmogrification using the stats corresponding
// to the original equipment template.
QRY
QRY_WW_CE_UsingStatsFromTemplate((CHARACTER)_Character, (GUIDSTRING)_Template)
AND
DB_WW_TL_TransmogrifiedArmors(_Item, _, _Template, _)
AND
DB_WW_TL_EquippedTransmogrifications((GUIDSTRING)_Character, (ITEM)_Item)
THEN
DB_NOOP(1);

//END_REGION
